function initGameAndCinematic() { // 1. Ocultar la pantalla de inicio con transición const startOverlay = document.getElementById('start-overlay'); if (startOverlay) { startOverlay.style.opacity = '0'; setTimeout(() => { startOverlay.style.display = 'none'; }, 500); } // 2. Hacer aparecer la cinemática con un fundido (Fade In) const container = document.getElementById('cinematic-container'); if (container) { container.style.opacity = '1'; } // 3. Reproducir el audio (al haber un toque de usuario previo, el celular lo permite) const introAudio = new Audio('/intro-cinematica.mp3'); introAudio.volume = 0.9; introAudio.play().catch(e => { console.log("Error al reproducir audio:", e); }); // 4. Activar partículas y cinemática spawnParticles(); // 5. Ocultar la cinemática tras 4.2 segundos (Fade Out) y pasar al login setTimeout(() => { if (container) { container.style.opacity = '0'; setTimeout(() => { container.style.display = 'none'; }, 800); } }, 4200); }
